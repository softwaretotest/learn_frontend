<CSSEditor 
  defaultHTML={`<div class="wrapper">
  <input id="input" class="sp-input-amount count-xx-large" value="123" />
</div>

<script>
  const input = document.getElementById("input");

  const getFontsizer = (value) => {
    const length = String(value).length;

    return length >= 13
      ? "count-xx-small"
      : length >= 11
      ? "count-x-small"
      : length >= 9
      ? "count-smaller"
      : length >= 7
      ? "count-small"
      : length >= 5
      ? "count-medium"
      : length >= 3
      ? "count-x-large"
      : "count-xx-large";
  };

  input.addEventListener("input", (e) => {
    const value = e.target.value;

    input.className = "sp-input-amount " + getFontsizer(value);
  });
</script>`}

defaultCSS={`:root {
