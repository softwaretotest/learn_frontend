<script>
  const getKeywordFont = (value) => {
    const length = value.length;

    return length >= 13
      ? "count-xx-small"
      : length >= 11
        ? "count-x-small"
        : length >= 9
          ? "count-smaller"
          : length >= 7

  const getRemFont = (value) => {
    const length = value.length;

    return length >= 13
      ? "rem-xx-small"
      : length >= 11
        ? "rem-x-small"
        : length >= 9
          ? "rem-smaller"
          : length >= 7
            ? "rem-small"
            : length >= 5
              ? "rem-medium"
              : length >= 3
                ? "rem-x-large"
                : "rem-xx-large";
  };

  const checkOverflow = (el) => {
    return el.scrollWidth > el.clientWidth;
  };

  const bind = (id, mapper) => {
    const input = document.getElementById(id);

    input.addEventListener("input", (e) => {
      const value = e.target.value;

      input.className = "sp-input " + mapper(value);

      if (checkOverflow(input)) {
        input.classList.add("overflow");
      } else {
        input.classList.remove("overflow");
      }
    });
  };

  bind("remInput", getRemFont);
  bind("keywordInput", getKeywordFont);
</script>`}
